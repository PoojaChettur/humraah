class SplashScreen extends StatefulWidget {
  const SplashScreen({super.key});

  @override
  _SplashScreenState createState() => _SplashScreenState();
}

class _SplashScreenState extends State<SplashScreen> with TickerProviderStateMixin {
  late AnimationController _iconController;
  late Animation<double> _rotation;

  @override
  void initState() {
    super.initState();
    _iconController = AnimationController(vsync: this, duration: const Duration(seconds: 4));
    _rotation = Tween<double>(begin: 0, end: 2 * pi).animate(_iconController);
    _iconController.repeat();

    Timer(const Duration(seconds: 4), () {
      Navigator.pushReplacement(context, MaterialPageRoute<void>(builder: (_) => SignInScreen()));
    });
  }

  @override
  void dispose() {
    _iconController.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: Colors.black,
      body: Stack(
        children: [
          Center(
            child: Column(
              mainAxisAlignment: MainAxisAlignment.center,
              children: [
                AnimatedBuilder(
                  animation: _rotation,
                  builder: (_, Widget? child) {
                    return Transform.rotate(
                      angle: _rotation.value,
                      child: const Icon(Icons.location_city, size: 80, color: Colors.cyanAccent),
                    );
                  },
                ),
                const SizedBox(height: 20),
                ShaderMask(
                  shaderCallback: (Rect bounds) => const LinearGradient(
                    colors: [Colors.cyanAccent, Colors.greenAccent],
                    tileMode: TileMode.mirror,
                  ).createShader(bounds),
                  child: const Text(
                    'Humraah',
                    style: TextStyle(
                      fontSize: 40,
                      fontWeight: FontWeight.bold,
                      color: Colors.white,
                      letterSpacing: 2,
                    ),
                  ),
                ),
                const SizedBox(height: 10),
                Text(
                  'Smart. Civic. Connected.',
                  style: TextStyle(
                    fontSize: 18,
                    color: Colors.grey[300],
                    fontStyle: FontStyle.italic,
                  ),
                ),
                const SizedBox(height: 40),
                const ShimmerBar(),
              ],
            ),
          ),
        ],
      ),
    );
  }
}

class ShimmerBar extends StatefulWidget {
  const ShimmerBar({super.key});

  @override
  _ShimmerBarState createState() => _ShimmerBarState();
}

class _ShimmerBarState extends State<ShimmerBar> with SingleTickerProviderStateMixin {
  late AnimationController _controller;
  late Animation<double> _animation;

  @override
  void initState() {
    super.initState();
    _controller = AnimationController(vsync: this, duration: const Duration(seconds: 2))..repeat();
    _animation = Tween<double>(begin: 0, end: 1).animate(_controller);
  }

  @override
  void dispose() {
    _controller.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return AnimatedBuilder(
      animation: _animation,
      builder: (_, Widget? __) {
        return Container(
          width: 200,
          height: 8,
          decoration: BoxDecoration(
            gradient: LinearGradient(
              colors: const [Colors.cyanAccent, Colors.greenAccent],
              stops: <double>[_animation.value, (_animation.value + 0.2).clamp(0.0, 1.0)],
              begin: Alignment.centerLeft,
              end: Alignment.centerRight,
            ),
            borderRadius: BorderRadius.circular(4),
          ),
        );
      },
    );
  }
}

class SignInScreen extends StatelessWidget {

  @override

    return Scaffold(

            mainAxisAlignment: MainAxisAlignment.center,

              Image.network(
                'https://www.gstatic.com/flutter-onestack-prototype/genui/example_1.jpg',
                height: 120,
                width: 120,

                controller: emailCtrl,
                decoration: const InputDecoration(
                  labelText: 'Email',

              const SizedBox(height: 15),
              TextField(

class ComplaintForm extends StatefulWidget {

class _ComplaintFormState extends State<ComplaintForm> {
  String selectedLanguage = 'English';
  String selectedCategory = 'Road Issue';
  String description = '';
  String location = '';
  String photoStatus = 'No photo uploaded';
  String speechInput = 'Speech-to-text not activated';

  @override

    return Scaffold(

        child: ListView(
          children: [
            Text('Language', style: TextStyle(color: Theme.of(context).colorScheme.primary)),
            DropdownButton<String>(
              isExpanded: true,
              value: selectedLanguage,
              dropdownColor: Colors.grey[800],
              style: const TextStyle(color: Colors.white),
              iconEnabledColor: Colors.white70,
              items: ['English', 'Hindi', 'Tamil', 'Telugu']
                  .map<DropdownMenuItem<String>>((String lang) => DropdownMenuItem<String>(
                      value: lang, child: Text(lang)))

                if (val != null) {
                  setState(() => selectedLanguage = val);

            Text('Speech Input', style: TextStyle(color: Theme.of(context).colorScheme.primary)),
